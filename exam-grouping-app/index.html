1. อ่านข้อมูลจาก Google Sheet (ตั้งค่าลิงก์ของคุณไว้ให้แล้ว)

ระบบอ่านคอลัมน์ตามที่คุณกำหนด: A=รหัสผู้สมัคร, D+E=ชื่อ-นามสกุลนักเรียน (F=ชื่อเล่น), L=โรงเรียนเดิม, S=ผู้ปกครอง, Y=อาชีพ/สถานที่ทำงาน

2. บันทึกผลการจัดกลุ่มและสีเก็บไว้บน Google Sheet ออนไลน์ (ไม่เปลืองพื้นที่เครื่อง)

วาง Web App URL (จาก Google Apps Script) ด้านล่าง เพื่อให้ทุกครั้งที่ลากจัดกลุ่มหรือเปลี่ยนสี ระบบบันทึกเก็บไว้ในชีตชื่อ AppState บน Google Sheet ของคุณโดยอัตโนมัติ เปิดจากเครื่องไหนผ่าน GitHub Pages ข้อมูลก็ตรงกันเสมอ:

วิธีทำ (ใช้เวลา 1 นาที): เปิด Google Sheet ของคุณ > ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script > ลบโค้ดเดิมแล้ววางโค้ดด้านล่างนี้ > กดปุ่ม การทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment) > เลือกประเภทเป็น เว็บแอป (Web app) > ช่องผู้มีสิทธิ์เข้าถึงเลือก "ทุกคน (Anyone)" > กด Deploy แล้วคัดลอก Web App URL มาวางในช่องด้านบนครับ 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'readSheet';
  if (action === 'loadState') {
    var stateSheet = ss.getSheetByName('AppState');
    var stateJson = stateSheet ? stateSheet.getRange('A1').getValue() : '';
    return ContentService.createTextOutput(stateJson || '{}').setMimeType(ContentService.MimeType.JSON);
  }
  var sheet = ss.getSheets()[0];
  var data = sheet.getDataRange().getDisplayValues();
  return ContentService.createTextOutput(JSON.stringify({ rows: data })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var stateSheet = ss.getSheetByName('AppState') || ss.insertSheet('AppState');
  stateSheet.getRange('A1').setValue(e.postData.contents);
  stateSheet.getRange('B1').setValue('อัปเดตล่าสุด: ' + new Date().toLocaleString('th-TH'));
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
